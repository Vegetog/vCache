cmake_minimum_required(VERSION 3.16)
project(vCache VERSION 0.1.0 LANGUAGES CXX)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

find_package(Threads REQUIRED)

add_library(vcache INTERFACE)
add_library(vcache::vcache ALIAS vcache)
target_compile_features(vcache INTERFACE cxx_std_17)
target_include_directories(vcache INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(vcache INTERFACE Threads::Threads)

option(VCACHE_BUILD_BENCHMARKS "Build reproducible cache benchmarks" ON)
option(VCACHE_BUILD_TESTS "Build correctness tests" ON)
option(VCACHE_ENABLE_ASAN "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(VCACHE_ENABLE_TSAN "Enable ThreadSanitizer" OFF)
if(VCACHE_ENABLE_ASAN AND VCACHE_ENABLE_TSAN)
    message(FATAL_ERROR "ASan and TSan must use separate build directories")
endif()
if(VCACHE_ENABLE_ASAN OR VCACHE_ENABLE_TSAN)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        message(FATAL_ERROR "Sanitizer builds require Clang or GCC")
    endif()
    if(VCACHE_ENABLE_ASAN)
        set(VCACHE_SANITIZER address,undefined)
    else()
        set(VCACHE_SANITIZER thread)
    endif()
    target_compile_options(vcache INTERFACE -fsanitize=${VCACHE_SANITIZER} -fno-omit-frame-pointer)
    target_link_options(vcache INTERFACE -fsanitize=${VCACHE_SANITIZER})
endif()

add_executable(vcache_example examples/basic.cpp)
target_link_libraries(vcache_example PRIVATE vcache::vcache)

if(VCACHE_BUILD_BENCHMARKS)
    add_subdirectory(benchmark)
endif()

if(VCACHE_BUILD_TESTS)
    enable_testing()
    add_executable(vcache_tests tests/cache_test.cpp)
    target_link_libraries(vcache_tests PRIVATE vcache::vcache)
    add_test(NAME cache_correctness COMMAND vcache_tests)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND AND VCACHE_BUILD_BENCHMARKS)
        add_test(NAME benchmark_tools COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_benchmark_tools.py
            --binary $<TARGET_FILE:vcache_bench>)
    endif()
endif()
